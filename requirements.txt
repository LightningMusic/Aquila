# Project Aquila - Python dependencies
#
# Core dependencies required by the code currently implemented under
# src/. As additional subsystems are built out, add their real
# dependencies here rather than assuming they're installed.

# config/ - YAML is the primary format for configs/*.yaml
# (src/config/loaders/yaml_loader.py).
pyyaml>=6.0

# bios/ - Windows firmware/hardware queries (WMI). Optional at
# runtime: every bios/providers/*.py call site that uses it is
# wrapped in a try/except ImportError and degrades to an
# honestly-unsupported result when it's absent, but install it on
# Windows deployment media for real firmware data collection.
wmi>=1.5.1; sys_platform == "win32"
pywin32>=306; sys_platform == "win32"

# api/ - src/api/client.py and src/api/download.py both import
# `requests` directly (HTTP session management, downloads). This was
# missing from requirements.txt even though api/ already depended on
# it -- a real gap found while building bootstrap/, which reuses
# api.client.ApiClient for Deployment Controller communication.
# Fixed here at the root rather than worked around in a caller.
requests>=2.31.0
